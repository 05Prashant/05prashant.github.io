const fs = require('fs');

const dir = 'src/assets/work/docker-onboarding';
fs.mkdirSync(dir, { recursive: true });

  .jpeg({ quality: 80 })
  .toFile(path.join(dir, 'hero.jpg'));

await sharp({ create: { width: 800, height: 600, channels: 3, background: color } })
  .jpeg({ quality: 80 })
  .toFile(path.join(dir, 'card-main.jpg'));
