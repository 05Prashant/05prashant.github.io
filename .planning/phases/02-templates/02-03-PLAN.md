**Full file content:**

```astro
---
import { getCollection, render } from 'astro:content';
import { Image } from 'astro:assets';
import BaseLayout from '../../layouts/BaseLayout.astro';

export async function getStaticPaths() {
  const entries = await getCollection('work', ({ data }) => !data.draft);
  return entries.map((entry) => ({
    params: { id: entry.id },
    props: { entry },
  }));
}

const { entry } = Astro.props;
const { Content } = await render(entry);
---

<BaseLayout title={entry.data.title} description={entry.data.description}>
```

Critical implementation notes:
- `render` is imported from `'astro:content'` as a named import alongside `getCollection` — NOT called as `entry.render()` (Astro 4 pattern, removed in Astro 5)
- `params: { id: entry.id }` — the key `id` must match the filename bracket `[id]`
- `entry.id` (not `entry.slug`) — `slug` no longer exists on Astro 5 collection entries; using it causes a TypeScript error
- `getStaticPaths` filters `!data.draft` — draft entries get no page generated
- `<Content />` renders the full compiled markdown body — everything below the frontmatter in the `.md` file
